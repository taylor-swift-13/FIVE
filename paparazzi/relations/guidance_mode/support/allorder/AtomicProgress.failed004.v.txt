Lemma atomic_assign_flat: forall ge e m l r ty,
 atomic_eval ge e m(Eassign l r ty)=flat_eval ge e m(Eassign l r ty).
Proof. intros;unfold atomic_eval,flat_eval,commit.
 destruct(type_eq (typeof l)ty);auto.
 destruct(pure_lval ge e m l) as [[[b ofs]bf]|];auto.
 Qed.

Lemma assignop_data: forall ge e m op l r tr ty v mf,
 atomic_expr(Eassignop op l r tr ty)=true ->
 atomic_eval ge e m(Eassignop op l r tr ty)=Some(v,mf) ->
 exists b ofs bf vl vr,
 pure_lval ge e m l=Some(b,ofs,bf) /\
 pure_rval ge e m r=Some vr /\
 deref_loc ge(typeof l)m b ofs bf E0 vl.
Proof.
 intros ge e m op l r tr ty v mf AT EV.
 simpl in AT;core_unpack.
 unfold atomic_eval in EV;option_cases EV.
 match goal with D:do_deref_loc _ _ _ _ _ _ _=Some _ |- _=>
 destruct(do_deref_loc_sound _ _ _ _ _ _ _ _ _ _ D) as [DR PT]
 end.
 assert(TT:t=E0) by(eapply deref_loc_silent;eauto);subst t.
 eauto 12.
Qed.

Lemma assignop_immediate: forall ge e m op l r tr ty v mf,
 atomic_expr(Eassignop op l r tr ty)=true ->
 atomic_eval ge e m(Eassignop op l r tr ty)=Some(v,mf) ->
 imm_safe ge e RV(Eassignop op l r tr ty)m.
Proof.
 intros ge e m op l r tr ty v mf AT EV.
 pose proof(assignop_data ge e m op l r tr ty v mf AT EV) as [b[ofs[bf[vl[vr[LL[RR LOAD]]]]]]].
 simpl in AT;core_unpack.
 pose proof(proj2(pure_eval_big l ge e m ltac:(assumption)) _ _ _ LL) as BL.
 pose proof(proj1(pure_eval_big r ge e m ltac:(assumption)) _ RR) as BR.
 destruct(literal LV l) eqn:LITL.
 - destruct(literal_l_inv _ LITL) as [bl[ol[bfl[tl LE]]]];subst l.
   invert_literal_eval;cbn[typeof] in *;subst.
   destruct(literal RV r) eqn:LITR.
   + destruct(literal_r_inv _ LITR) as [vr0[tr0 RE]];subst r.
     invert_literal_eval;cbn[typeof] in *;subst.
     eapply imm_safe_rred with(C:=fun x=>x);[|constructor].
     unfold atomic_eval in EV;cbn[typeof pure_lval pure_rval] in EV;destruct(type_eq tl ty);try discriminate.
     subst ty;eapply red_assignop;exact LOAD.
   + eapply lift_immediate with(from:=RV)(C:=fun x=>Eassignop op _ x tr ty);eauto.
     constructor;constructor.
     eapply(proj1(big_immediate ge e m));exact BR.
 - eapply lift_immediate with(from:=LV)(C:=fun x=>Eassignop op x r tr ty);eauto.
   constructor;constructor.
   eapply(proj2(big_immediate ge e m));exact BL.
Qed.

Lemma atomic_not_stuck: forall ge e m a v mf,
 atomic_expr a=true -> atomic_eval ge e m a=Some(v,mf) -> Csem.not_stuck ge e a m.
Proof.
 intros ge e m a v mf AT EV.
 destruct a;try solve[
 rewrite atomic_eval_simple in EV by assumption;
 destruct(pure_rval ge e m _)eqn:E;try discriminate;
 eapply pure_not_stuck;eauto].
 - rewrite atomic_assign_flat in EV.
   eapply flat_not_stuck;[simpl in *;core_unpack;core_build;eauto|exact EV].
 - pose proof(assignop_data _ _ _ _ _ _ _ _ _ _ AT EV) as [b[ofs[bf[vl[vr[LL[RR LOAD]]]]]]].
   simpl in AT;core_unpack.
   intros k C sub CT EQ;inversion CT;subst;try discriminate.
   + cbn in EQ;subst sub;eapply assignop_immediate;eauto;simpl;core_build;eauto.
   + inversion EQ;subst.
     pose proof(proj2(pure_eval_big _ ge e m ltac:(assumption)) _ _ _ LL) as BE.
     match goal with CT:Csem.context _ LV _ |- _=>
      pose proof(big_context_hole _ _ _ CT sub ge e m ltac:(assumption)
        ltac:(unfold big_defined;eauto)) as HS end.
     destruct k;unfold big_defined in HS.
     * destruct HS as [bb[oo[bff HO]]];eapply(proj2(big_immediate ge e m));eauto.
     * destruct HS as [w HO];eapply(proj1(big_immediate ge e m));eauto.
   + inversion EQ;subst.
     pose proof(proj1(pure_eval_big _ ge e m ltac:(assumption)) _ RR) as BE.
     match goal with CT:Csem.context _ RV _ |- _=>
      pose proof(big_context_hole _ _ _ CT sub ge e m ltac:(assumption)
        ltac:(unfold big_defined;eauto)) as HS end.
     destruct k;unfold big_defined in HS.
     * destruct HS as [bb[oo[bff HO]]];eapply(proj2(big_immediate ge e m));eauto.
     * destruct HS as [w HO];eapply(proj1(big_immediate ge e m));eauto.
Qed.
